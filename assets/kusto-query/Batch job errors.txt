Batch job errors:


let _batchJobId = ''; exceptions | where     timestamp < ago(30m) | extend    CustomDimensionsParsed = parse_json(customDimensions) | extend    ExecutionMode   = tostring(CustomDimensionsParsed.ExecutionMode) | extend    BatchJobId      = tostring(CustomDimensionsParsed.BatchJobId) | extend    BatchJobCaption = tostring(CustomDimensionsParsed.BatchJobCaption) | extend    ActivityId      = tostring(CustomDimensionsParsed.activityId) | extend    MethodName      = tostring(CustomDimensionsParsed.MethodName) | extend    ClassName       = tostring(CustomDimensionsParsed.ClassName) | extend    CallStack       = tostring(CustomDimensionsParsed.CallStack) | where     ExecutionMode   == "Batch" | where     isempty(_batchJobId) or tostring(BatchJobId) in (_batchJobId) | project   TimeStamp = timestamp,BatchJobId, BatchJobCaption, OuterMessage = outerMessage  | limit 30

Batch Excpetions:
customEvents | where timestamp between (ago(1h) .. now()) | where     name in ("BatchTaskFailure")  | extend    CustomDimensionsParsed = parse_json(customDimensions)  | extend    BatchJobCaption     = tostring(CustomDimensionsParsed.BatchJobCaption)  | extend    ActivityId          = tostring(CustomDimensionsParsed.activityId)  | extend    ClassName           = iif((tostring(CustomDimensionsParsed.ClassName) == "<empty>"), "", tostring(CustomDimensionsParsed.ClassName))  | extend    BatchJobId          = tostring(CustomDimensionsParsed.BatchJobId)  | extend    BatchTaskId         = tostring(CustomDimensionsParsed.BatchTaskId)  | extend    EventMessage        = tostring(CustomDimensionsParsed.EventMessage)  | extend    ExceptionType       = tostring(CustomDimensionsParsed.ExceptionType)  | extend    ExceptionMessage    = tostring(CustomDimensionsParsed.ExceptionMessage)  | extend    CallStack           = tostring(CustomDimensionsParsed.CallStack)  | project   TimeStamp = timestamp, RoleInstance = cloud_RoleInstance, ActivityId, BatchJobCaption, ClassName, BatchJobId, BatchTaskId, EventMessage, ExceptionType, ExceptionMessage, CallStack


Batch Execution history:
customEvents | where timestamp between (ago(15m) .. now()) | where     name in ("BatchTaskStart","BatchTaskFinished","BatchTaskFailure")  | extend    CustomDimensionsParsed = parse_json(customDimensions)   | extend    InfoMessageParsed   = parse_json(tostring(CustomDimensionsParsed.InfoMessage))    | extend    ActivityId          = tostring(CustomDimensionsParsed.activityId)    | extend    ClassName           = tostring(CustomDimensionsParsed.ClassName)    | extend    BatchJobId          = tostring(CustomDimensionsParsed.BatchJobId)    | extend    BatchJobTaskId1      = tostring(CustomDimensionsParsed.BatchJobTaskId)    | extend    BatchJobTaskId2      = tostring(CustomDimensionsParsed.BatchTaskId)    | extend    BatchJobTaskId      = iif(isnotempty(BatchJobTaskId1), BatchJobTaskId1, BatchJobTaskId2)      | extend Caption = tostring(CustomDimensionsParsed.BatchJobCaption)      | extend    StartTime           = iff(name == "BatchTaskStart"    , timestamp, datetime(null))      | extend    EndTime             = iff(name == "BatchTaskFinished"   , timestamp, datetime(null))       | extend    ErrorTime           = iff(name == "BatchTaskFailure"   , timestamp, datetime(null))        | extend    RetryCount          = iff(name == "BatchTaskStart"    , InfoMessageParsed.RetryCount, "")        | project   StartTime, EndTime,ErrorTime,RetryCount, ActivityId, ClassName, BatchJobId, RoleInstance = cloud_RoleInstance, BatchJobTaskId,Caption      | summarize   StartTime         = min(StartTime),               CompletionTime    = max(EndTime),               ErrorTime         = max(ErrorTime),               RetryCount        = take_any(RetryCount),               ClassName         = any(ClassName),                BatchJobId        = any(BatchJobId),               BatchJobTaskId    = any(BatchJobTaskId),               RoleInstance      = any(RoleInstance),               Caption = any(Caption) by ActivityId | where isnotempty(StartTime) | extend BatchAOS = strcat(split(RoleInstance, "-")[0]) | project ActivityId, BatchAOS, BatchJobId,Caption, BatchJobTaskId, ClassName, StartTime, CompletionTime, ElapsedTime = CompletionTime - StartTime, RetryCount

Batch Throttling:
customEvents  | where     timestamp between (ago(1h) .. now()) | where     name == "BatchThrottled"  | extend    customDimensionsParsed = parse_json(customDimensions) | extend    BatchJobId          = customDimensionsParsed.BatchJobId | extend    EventId             = customDimensionsParsed.eventId  | extend    EnvironmentId       = customDimensionsParsed.environmentId  | extend    LegalEntity         = customDimensionsParsed.LegalEntity  | extend    ActivityId          = customDimensionsParsed.activityId  | extend    EventMessage        = customDimensionsParsed.EventMessage  | extend    InfoMessage         = customDimensionsParsed.InfoMessage  | extend    InfoMessageParsed       = parse_json(tostring(InfoMessage))  | extend    CurrentMachineCPU       = round(toreal(InfoMessageParsed.CurrentMachineCpu), 0)  | extend    CurrentMachineMemory    = InfoMessageParsed.CurrentMachineMemory  | extend    CurrentSqlDtu           = InfoMessageParsed.CurrentSqlDtu  | project   timestamp, EventId, name, user_Id, session_Id, cloud_RoleInstance, EnvironmentId, LegalEntity, ActivityId, EventMessage, CurrentMachineCPU, CurrentMachineMemory, CurrentSqlDtu
